## ⚡ 1. The Mobile Performance Budget: 60 FPS vs. 120 FPS

The human eye perceives UI stutter (jank) whenever frame delivery misses the hardware VSYNC deadline:
- **60 Hz Displays:** Frame budget = **$16.6\text{ms}$**
- **120 Hz Displays (ProMotion / Smooth Display):** Frame budget = **$8.33\text{ms}$**

```
===================================================================================================
                                  FRAME RENDERING TIMELINE BUDGET
===================================================================================================

  0ms                             8.33ms (120Hz Target)              16.6ms (60Hz Target)
  ├─────────────────────────────────┼─────────────────────────────────┤
  │ [ Measure ] ──► [ Layout ] ──► [ Draw ] ──► [ GPU Render ]        │ ===> SMOOTH FRAME
  └─────────────────────────────────┴─────────────────────────────────┘

  0ms                                                                16.6ms
  ├───────────────────────────────────────────────────────────────────┼────────────► (JANK!)
  │ [ Heavy JSON Parsing on Main Thread: 24ms ]                       │ Frame Dropped!
  └───────────────────────────────────────────────────────────────────┴────────────► User sees stutter.
```

### 1.1 Cold-Start Optimization & Baseline Profiles
Android app cold start traverses three phases:
1. OS boots process and loads DEX bytecode into memory.
2. `Application.onCreate()` initializes SDKs (Firebase, Hilt, Analytics).
3. First Activity inflates and renders initial frame.

**Baseline Profiles:** By pre-compiling critical user journeys (app startup, catalog scrolling) using profile-guided optimization Ahead-of-Time (AOT), **Baseline Profiles improve app launch speeds by up to 30–40%!**

## 🛡️ 2. Hardware-Backed Security: The Android Keystore System

Never store cryptographic keys or plaintext JWT tokens in SharedPreferences or raw files on disk. Rooted devices can inspect private app directories (`/data/data/<package>/`).

The **Android Keystore System** stores cryptographic key material inside a **Hardware Security Module (HSM)** or **Trusted Execution Environment (TEE)**:
- Key material **never enters application memory**.
- Cryptographic operations (AES encryption, RSA signing) execute inside the isolated hardware enclave.

```
===================================================================================================
                                 ANDROID KEYSTORE ENCLAVE FLOW
===================================================================================================

  [ Android App Process ]                           [ Hardware Security Module (TEE / StrongBox) ]
             │                                                                   │
             ├── 1. Cipher.init(ENCRYPT, key) ──────────────────────────────────►│
             ├── 2. Cipher.doFinal(plaintext) ──────────────────────────────────►│ (Key never leaves TEE!)
             │                                                                   │ (Encrypts in hardware)
             │◄── 3. Returns Ciphertext + Initialization Vector (IV) ─────────────┤
```

    App["Application Process<br/>(User Realm)"] --> KeyStore["Android KeyStore Provider"]

### 3.1 Android Implementation: Hardware Keystore AES-256 GCM Encryption

```kotlin
package com.enterprise.mobile.security

import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import java.security.KeyStore
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec
import android.util.Base64

// Production Hardware-Backed Secure Enclave Manager
class SecureEnclaveStorage {

    private val KEY_ALIAS = "EnterpriseAuthMasterKey"
    private val ANDROID_KEYSTORE = "AndroidKeyStore"
    private val TRANSFORMATION = "AES/GCM/NoPadding"

    private val keyStore: KeyStore = KeyStore.getInstance(ANDROID_KEYSTORE).apply {
        load(null)
    }

    init {
        createMasterKeyIfNotExists()
    }

    private fun createMasterKeyIfNotExists() {
        if (!keyStore.containsAlias(KEY_ALIAS)) {
            val keyGenerator = KeyGenerator.getInstance(
                KeyProperties.KEY_ALGORITHM_AES,
                ANDROID_KEYSTORE
            )

            val spec = KeyGenParameterSpec.Builder(
                KEY_ALIAS,
                KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
            )
                .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                .setKeySize(256)
                .setUserAuthenticationRequired(false) // Set true for biometric-gated keys!
                .build()

            keyGenerator.init(spec)
            keyGenerator.generateKey() // Key generated directly inside hardware enclave!
        }
    }

    fun encrypt(plainText: String): EncryptedPayload {
        val secretKey = keyStore.getKey(KEY_ALIAS, null) as SecretKey
        val cipher = Cipher.getInstance(TRANSFORMATION)
        cipher.init(Cipher.ENCRYPT_MODE, secretKey)

        val iv = cipher.iv // 12-byte GCM Initialization Vector
        val cipherText = cipher.doFinal(plainText.toByteArray(Charsets.UTF_8))

        return EncryptedPayload(
            cipherTextBase64 = Base64.encodeToString(cipherText, Base64.NO_WRAP),
            ivBase64 = Base64.encodeToString(iv, Base64.NO_WRAP)
        )
    }

    fun decrypt(payload: EncryptedPayload): String {
        val secretKey = keyStore.getKey(KEY_ALIAS, null) as SecretKey
        val cipher = Cipher.getInstance(TRANSFORMATION)
        val iv = Base64.decode(payload.ivBase64, Base64.NO_WRAP)
        val spec = GCMParameterSpec(128, iv) // 128-bit authentication tag

        cipher.init(Cipher.DECRYPT_MODE, secretKey, spec)
        val decryptedBytes = cipher.doFinal(Base64.decode(payload.cipherTextBase64, Base64.NO_WRAP))

        return String(decryptedBytes, Charsets.UTF_8)
    }
}

data class EncryptedPayload(val cipherTextBase64: String, val ivBase64: String)
```

### 3.2 Flutter Implementation: Hardware-Backed Secure Storage & Root Detection

```dart
import 'dart:io';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class SecureVaultService {
  // Configured to mandate Android Hardware Keystore encrypted shared preferences
  static const _storage = FlutterSecureStorage(
    aOptions: AndroidOptions(
      encryptedSharedPreferences: true,
      resetOnError: true,
    ),
  );

  static Future<void> saveToken(String token) async {
    await _storage.write(key: 'auth_jwt_token', value: token);
  }

  static Future<String?> getToken() async {
    return await _storage.read(key: 'auth_jwt_token');
  }

  // Production Root Detection Check
  static bool isDeviceCompromised() {
    if (Platform.isAndroid) {
      final suPaths = [
        '/system/app/Superuser.apk',
        '/sbin/su',
        '/system/bin/su',
        '/system/xbin/su',
        '/data/local/xbin/su',
        '/data/local/bin/su',
        '/system/sd/xbin/su',
        '/system/bin/failsafe/su',
        '/data/local/su',
      ];

      for (final path in suPaths) {
        if (File(path).existsSync()) {
          print('SECURITY ALERT: Root binary found at $path!');
          return true;
        }
      }
    }
    return false;
  }
}
```
